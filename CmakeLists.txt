cmake_minimum_required(VERSION 3.15)
project(rtneuralCsound VERSION 0.1.0)

add_subdirectory(vendor/RTNeural)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

function(add_tool target_name)
    add_executable(${target_name} src/${target_name}/${target_name}.cpp)
    target_include_directories(${target_name} PRIVATE ${CMAKE_SOURCE_DIR}/vendor)
    target_link_libraries(${target_name} PRIVATE RTNeural)
    target_compile_features(${target_name} PRIVATE cxx_std_17)
endfunction()

add_tool(process_wav)
add_tool(process_wav_torch)
add_tool(process_wav_torch_param)

# Use headers from the installed Csound to ensure struct layout matches the binary.
# The DLL must be compiled against the same Csound version it will run with. 
# Mismatched headers cause silent opcode registration failure (wrong CSOUND struct layout).
# Default path is the standard Csound 7 Windows install. On Mac/Linux, override with
# -DCSOUND_INCLUDE_DIR=/usr/local/include/csound or wherever your install put the headers.
set(CSOUND_INCLUDE_DIR "C:/Program Files/Csound7/include/csound" CACHE PATH "Csound include directory")

add_library(moognn SHARED src/csound_opcode/moognn.cpp)
target_include_directories(moognn PRIVATE
    ${CSOUND_INCLUDE_DIR}
    ${CMAKE_SOURCE_DIR}/vendor/RTNeural
)
target_link_libraries(moognn PRIVATE RTNeural)
target_compile_features(moognn PRIVATE cxx_std_17)
set_target_properties(moognn PROPERTIES PREFIX "")
